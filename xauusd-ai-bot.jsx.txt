const SYMBOL = "XAUUSD";
const BASE_GOLD = 2345.80;

function genCandles(n = 200) {
  let p = BASE_GOLD;
  return Array.from({ length: n }, (_, i) => {
    const v = p * 0.0025;
    const o = p + (Math.random() - .5) * v;
    const c = o + (Math.random() - .5) * v * 2;
    const h = Math.max(o, c) + Math.random() * v * .7;
    const l = Math.min(o, c) - Math.random() * v * .7;
    p = c;
    return { t: Date.now() - (n - i) * 60000, o, h, l, c, vol: 300 + Math.random() * 1400 };
  });
}

// ══════════════════════════════════════════════════════════════
// INDICATORS (mirrors Python code)
// ══════════════════════════════════════════════════════════════
function computeATR(candles, period = 14) {
  return candles.map((_, i) => {
    if (i < period) return candles[i].h - candles[i].l;
    const sl = candles.slice(i - period, i);
    return sl.reduce((s, c) => s + (c.h - c.l), 0) / period;
  });
}

function ema(arr, period) {
  const k = 2 / (period + 1);
  const out = [arr[0]];
  for (let i = 1; i < arr.length; i++) out.push(arr[i] * k + out[i - 1] * (1 - k));
  return out;
}

function computeRSI(closes, period = 14) {
  return closes.map((_, i) => {
    if (i < period) return 50;
    let g = 0, l = 0;
    for (let j = i - period + 1; j <= i; j++) {
      const d = closes[j] - closes[j - 1];
      d > 0 ? g += d : l -= d;
    }
    return 100 - 100 / (1 + g / (l || .0001));
  });
}

// ══════════════════════════════════════════════════════════════
// REGIME DETECTOR  (Python: detect_regime)
// ══════════════════════════════════════════════════════════════
function detectRegime(candles) {
  const closes = candles.map(c => c.c);
  const atrArr = computeATR(candles);
  const fast = ema(closes, 20);
  const slow = ema(closes, 50);
  const n = closes.length - 1;
  const slope = fast[n] - slow[n];
  const atrNow = atrArr[n];
  const atrMean = atrArr.reduce((s, v) => s + v, 0) / atrArr.length;
  if (slope > 0 && atrNow > atrMean) return "TREND_UP";
  if (slope < 0 && atrNow > atrMean) return "TREND_DOWN";
  return "RANGE";
}

// ══════════════════════════════════════════════════════════════
// LIQUIDITY SWEEP DETECTOR (Python: detect_liquidity_sweep)
// ══════════════════════════════════════════════════════════════
function detectLiquiditySweep(candles) {
  const n = candles.length - 1;
  const prev = candles[n - 1], last = candles[n];
  if (last.l < prev.l && last.c > prev.l) return "SWEEP_BUY";
  if (last.h > prev.h && last.c < prev.h) return "SWEEP_SELL";
  return "NO_SWEEP";
}

// ══════════════════════════════════════════════════════════════
// MOMENTUM CONFIRMATION  (Python: momentum_confirmation)
// ══════════════════════════════════════════════════════════════
function momentumConfirmation(candles) {
  const c = candles[candles.length - 1];
  const body = Math.abs(c.c - c.o);
  const range = c.h - c.l;
  return range > 0 && body / range > 0.60;
}

// ══════════════════════════════════════════════════════════════
// SCALPING SIGNAL  (Python: scalping_signal)
// ══════════════════════════════════════════════════════════════
function scalpingSignal(candles, regime) {
  if (!["TREND_UP", "TREND_DOWN"].includes(regime)) return "NO_TRADE";
  const sweep = detectLiquiditySweep(candles);
  const momentum = momentumConfirmation(candles);
  if (regime === "TREND_UP"   && sweep === "SWEEP_BUY"  && momentum) return "SCALP_BUY";
  if (regime === "TREND_DOWN" && sweep === "SWEEP_SELL" && momentum) return "SCALP_SELL";
  return "NO_TRADE";
}

// ══════════════════════════════════════════════════════════════
// XGBoost MODEL SIMULATOR (Python: SignalModel)
// ══════════════════════════════════════════════════════════════
function xgboostPredict(candles, regime) {
  const closes = candles.map(c => c.c);
  const atrArr = computeATR(candles);
  const fast = ema(closes, 20);
  const slow = ema(closes, 50);
  const n = closes.length - 1;
  const ret = (closes[n] - closes[n - 1]) / closes[n - 1];
  const atrNorm = atrArr[n] / closes[n];
  const emaDiff = (fast[n] - slow[n]) / closes[n];
  // Simulate feature weighting (tree ensemble logic approximation)
  let score = 0;
  score += emaDiff > 0 ? 1.2 : -1.2;
  score += ret > 0 ? 0.8 : -0.8;
  score += atrNorm > 0.002 ? 0.5 : -0.2;
  const noise = (Math.random() - .5) * 0.6;
  score += noise;
  const confidence = Math.min(0.95, 0.55 + Math.abs(score) * 0.07 + Math.random() * 0.08);
  const prediction = score > 0.4 ? 1 : score < -0.4 ? -1 : 0;
  return { prediction, confidence };
}

// ══════════════════════════════════════════════════════════════
// SIGNAL FUSION  (Python: generate_signal)
// ══════════════════════════════════════════════════════════════
function generateSignal(regime, prediction, confidence) {
  if (confidence < 0.60) return "NO_TRADE";
  if (regime === "TREND_UP"   && prediction === 1)  return "BUY";
  if (regime === "TREND_DOWN" && prediction === -1) return "SELL";
  if (regime === "RANGE")                           return "MEAN_REVERSION";
  return "NO_TRADE";
}

// ══════════════════════════════════════════════════════════════
// RISK ENGINE  (Python: risk_per_trade)
// ══════════════════════════════════════════════════════════════
function riskPerTrade(regime) {
  if (regime === "TREND_UP" || regime === "TREND_DOWN") return 0.02;
  if (regime === "RANGE") return 0.01;
  return 0.005;
}

function riskToLot(risk, balance = 10000) {
  const usdRisk = balance * risk;
  return Math.max(0.01, +(usdRisk / 1000).toFixed(2));
}

// ══════════════════════════════════════════════════════════════
// SESSION FILTER  (Python: trading_session_filter)
// ══════════════════════════════════════════════════════════════
function tradingSessionFilter() {
  const h = new Date().getUTCHours();
  return h >= 7 && h <= 20;
}

// ══════════════════════════════════════════════════════════════
// FULL BOT BRAIN  (Python: run_bot / autonomous_bot)
// ══════════════════════════════════════════════════════════════
function runBot(candles, balance) {
  const regime = detectRegime(candles);
  const { prediction, confidence } = xgboostPredict(candles, regime);
  const sweep = detectLiquiditySweep(candles);
  const scalp = scalpingSignal(candles, regime);
  const aiSignal = generateSignal(regime, prediction, confidence);
  const risk = riskPerTrade(regime);
  const lot = riskToLot(risk, balance);
  const inSession = tradingSessionFilter();
  const closes = candles.map(c => c.c);
  const atrArr = computeATR(candles);
  const n = closes.length - 1;
  const atr = atrArr[n];
  const price = closes[n];
  // Final fusion: scalping AND ai must agree (Python condition: signal != NO_TRADE && final_signal != NO_TRADE)
  let finalSignal = "NO_TRADE";
  if (inSession && scalp !== "NO_TRADE" && aiSignal !== "NO_TRADE") {
    finalSignal = scalp;
  }
  const sl = finalSignal.includes("BUY")  ? price - atr * 1.5 : finalSignal.includes("SELL") ? price + atr * 1.5 : null;
  const tp = finalSignal.includes("BUY")  ? price + atr * 3   : finalSignal.includes("SELL") ? price - atr * 3   : null;
  return { regime, prediction, confidence, sweep, scalp, aiSignal, finalSignal, risk, lot, atr, price, sl, tp, inSession };
}

function GoldChart({ candles, bot }) {
  const W = 640, H = 230;
  if (!candles || candles.length < 10) return null;
  const recent = candles.slice(-70);
  const closes = recent.map(c => c.c);
  const e20 = ema(closes, 20), e50 = ema(closes, 50);
  const mn = Math.min(...recent.map(c => c.l)), mx = Math.max(...recent.map(c => c.h)), rng = mx - mn || 1;
  const cw = W / recent.length;
  const py = v => H - ((v - mn) / rng) * (H - 22) - 11;
  const px = i => i * cw + cw * .5;
  const ep20 = e20.map((v, i) => `${i === 0 ? "M" : "L"}${px(i)},${py(v)}`).join(" ");
  const ep50 = e50.map((v, i) => `${i === 0 ? "M" : "L"}${px(i)},${py(v)}`).join(" ");
  return (
    <svg viewBox={`0 0 ${W} ${H}`} style={{ width: "100%", height: H }}>
      <defs>
        <linearGradient id="gg" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stopColor={C.gold} stopOpacity=".15" />
          <stop offset="100%" stopColor={C.gold} stopOpacity="0" />
        </linearGradient>
        <pattern id="dotgrid" width="30" height="20" patternUnits="userSpaceOnUse">
          <circle cx="1" cy="1" r=".4" fill={C.border} />
        </pattern>
      </defs>
      <rect width={W} height={H} fill="url(#dotgrid)" />
      {bot?.sl && <><line x1="0" y1={py(bot.sl)} x2={W} y2={py(bot.sl)} stroke={C.bear} strokeWidth=".9" strokeDasharray="4,4" opacity=".6" /><text x="4" y={py(bot.sl) - 3} fill={C.bear} fontSize="8" fontFamily="monospace">SL</text></>}
      {bot?.tp && <><line x1="0" y1={py(bot.tp)} x2={W} y2={py(bot.tp)} stroke={C.bull} strokeWidth=".9" strokeDasharray="4,4" opacity=".6" /><text x="4" y={py(bot.tp) - 3} fill={C.bull} fontSize="8" fontFamily="monospace">TP</text></>}
      {recent.map((c, i) => {
        const bull = c.c >= c.o, col = bull ? C.bull : C.bear;
        const bh = Math.max(1, Math.abs(py(c.o) - py(c.c)));
        return <g key={i}>
          <line x1={px(i)} y1={py(c.h)} x2={px(i)} y2={py(c.l)} stroke={col} strokeWidth=".9" opacity=".8" />
          <rect x={px(i) - cw * .38} y={py(Math.max(c.o, c.c))} width={cw * .76} height={bh} fill={col} opacity=".85" />
        </g>;
      })}
      <path d={ep20} fill="none" stroke={C.gold} strokeWidth="1.4" opacity=".85" />
      <path d={ep50} fill="none" stroke={C.purple} strokeWidth="1.4" opacity=".85" />
      <text x="6" y="13" fill={C.gold} fontSize="9" fontFamily="monospace">EMA20</text>
      <text x="55" y="13" fill={C.purple} fontSize="9" fontFamily="monospace">EMA50</text>
    </svg>
  );
}

// PIPELINE VISUALISER  (ANALYSER→CONFIRMER→ENTRER→GÉRER→SORTIR)
// ══════════════════════════════════════════════════════════════
const PIPELINE = ["ANALYSER", "CONFIRMER", "ENTRER", "GÉRER", "SORTIR"];
function Pipeline({ step }) {
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 0, marginBottom: 16 }}>
      {PIPELINE.map((s, i) => {
        const done = i <= step, active = i === step;
        return (
          <div key={i} style={{ display: "flex", alignItems: "center", flex: 1 }}>
            <div style={{
              flex: "none", width: 28, height: 28, borderRadius: "50%", display: "flex", alignItems: "center", justifyContent: "center",
              fontSize: 10, fontWeight: 700,
              background: done ? (active ? C.gold : C.bull + "33") : C.dim,
              border: `1.5px solid ${done ? (active ? C.gold : C.bull) : C.border}`,
              color: done ? (active ? C.bg : C.bull) : C.muted,
              boxShadow: active ? `0 0 14px ${C.gold}66` : "none",
              transition: "all .3s", fontSize: 9
            }}>{i + 1}</div>
            <div style={{ flex: 1, textAlign: "center", fontSize: 8, color: active ? C.gold : done ? C.bull : C.muted, transition: "color .3s", letterSpacing: .5 }}>{s}</div>
            {i < PIPELINE.length - 1 && <div style={{ width: 2, height: 2, borderRadius: "50%", background: i < step ? C.bull : C.border }} />}
          </div>
        );
      })}
    </div>
  );
}
